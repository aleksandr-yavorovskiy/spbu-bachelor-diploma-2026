\begin{frame}{Введение}
    \begin{itemize}
        \item Оцифровываются кириллические рукописи XIII века
        \item Возникает необходимость ручного редактирования полученного текста
        \item Сложности:
        \begin{itemize}
            \item Неконсистентность шрифтов
            \item Неудобство набора специальных кириллических символов
            \item Конвертация в PDF % TODO: убрать?
        \end{itemize}
        \item Необходимо создать приложение для удобной работы с кириллическими текстами
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Существующие решения}
    \begin{itemize}
        \item Word
        \begin{itemize}
            \item Де-факто стандарт
            \item Доступен экспорт в PDF
            \item Недостатки:
            \begin{itemize}
                \item Нет поддержки конвертации шрифтов: при смене шрифта символы могут \enquote{исчезнуть}
                \item Набор спецсимволов неудобен
                % \item TODO: частичная недоступность в РФ?
                % \item ??При конвертации в pdf все может поехать
            \end{itemize}
        \end{itemize}
        \item Overleaf
        \begin{itemize}
            \item Экспорт в PDF
            \item Использование TeX
            \item Недостатки:
            \begin{itemize}
                \item Компиляция лимитирована по объёму (при превышении лимита~--- платно)
                \item Нет поддержки конвертации шрифтов
                \item Нет встроенного инструмента для набора спецсимволов
                \item \enquote{Неблизко} обычному пользователю
            \end{itemize}
        \end{itemize}
        \item Прочие отечественные аналоги (Cyrillic Text Editor, Online Cyrillic Editor) не предназначены для работы с церковнославянскими текстами
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Существующие решения}
    Выводы:
    \begin{itemize}
        \item На данный момент специализированного приложения для работы с кириллическими и церковнославянскими текстами \textbf{нет}
        \item Основные недостатки существующих решений:
        \begin{itemize}
        \item Неудобный набор специфичных кириллических символов
        \item Не учитываются особенности шрифтов, отсутствует механизм конвертации
        \item Ограничения по объему текста в бесплатных версиях
        \end{itemize}
        \item Необходимо разработать инструмент для набора и редактирования кириллических текстов, автоматизирующий корректную конвертацию символов при смене шрифтов
    \end{itemize}
\end{frame}

        \item Спроектировать архитектуру приложения
        \item Реализовать разработанное решение
        \item Развернуть приложение на сервере

\begin{frame}{Обзор шрифтов}
    Рассматривалось 5 шрифтов:
    \begin{itemize}
        \item Buky Vede
        \item Flavius Universal и FlavExp
        \item Menaion Unicode
        \item Ponomar Unicode
    \end{itemize}

    Сравнение проводилось по 5 выделенным категориям:
    \begin{enumerate}
        \item Церковнославянский алфавит
        \item Дополнительные буквы из старославянского языка
        \item Выносные буквы (буквотитла)
        \item Диакритические знаки
        \item Знаки препинания
    \end{enumerate}

    \begin{center}
    \includegraphics[width=0.8\textwidth]{figures/compare_monograph_uk.png}
    \end{center}
\end{frame}
